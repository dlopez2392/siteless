import { timingSafeEqual } from 'node:crypto';
import { NextResponse } from 'next/server';
import { sql } from 'drizzle-orm';
import { env } from '@/env';
import { withCronRole } from '@/db/with-cron-role';
export const dynamic = 'force-dynamic';
export async function GET(request: Request) {
  const secret = env.CRON_SECRET;
  if (!secret) return NextResponse.json({ ok: false, reason: 'not_configured' }, { status: 503 });
  const got = Buffer.from(request.headers.get('authorization') ?? '');
  const want = Buffer.from(`Bearer ${secret}`);
  if (got.length !== want.length || !timingSafeEqual(got, want)) {
    return NextResponse.json({ ok: false, reason: 'unauthorized' }, { status: 401 });
  }
  try {
    const rows = await withCronRole((tx) => tx.execute(sql`select purged_org, purged_rows from app.purge_expired_place_coordinates('cron')`));
    // sum and count via rowsOf; return { ok: true, orgs, rowsPurged }
  } catch (e) {
    console.error('purge-places: purge failed', e instanceof Error ? e.name : 'unknown');
    return NextResponse.json({ ok: false, reason: 'purge_failed' }, { status: 500 });
  }
}
