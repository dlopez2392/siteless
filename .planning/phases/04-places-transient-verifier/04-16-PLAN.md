```ts
import 'server-only';
import { sql } from 'drizzle-orm';
import { db } from './client';
/** D-14 tier context. A workflow step has no Clerk request: the org comes from the run's own
 *  server-verified start input. Claims {o:{id}} ONLY — no `sub` (attribution falls to the
 *  app.actor_id GUC, like `etl:<script>`), no role claim (a workflow can never pass an admin-gated
 *  definer). Per transaction, never per step: both GUCs die at COMMIT. */
export async function withWorkerOrg<T>(clerkOrgId: string, actor: `workflow:${string}`, fn: (tx: Tx) => Promise<T>): Promise<T> {
  return db.transaction(async (tx) => {
    await tx.execute(sql`select set_config('request.jwt.claims', json_build_object('o', json_build_object('id', ${clerkOrgId}::text))::text, true)`);
    await tx.execute(sql`select set_config('app.actor_id', ${actor}, true)`);
    await tx.execute(sql.raw('set local role authenticated'));
    return fn(tx);
  });
}
```
