  throw new Error('scripts/db.ts: unknown target ' + target + ' (test | prod)');
}

const url = target === 'prod' ? process.env.SUPABASE_DB_URL : process.env.TEST_DATABASE_URL;
const varName = target === 'prod' ? 'SUPABASE_DB_URL' : 'TEST_DATABASE_URL';

// that reaches it would run the RLS suite's DDL against production.

if (target === 'test' && looksLikeSupabase) {
  throw new Error('scripts/db.ts: --target=test refuses a Supabase host. D-04.');
}
if (target === 'prod' && !looksLikeSupabase) {
  throw new Error('scripts/db.ts: --target=prod expects the Supabase session pooler URL.');
}
if (target === 'prod' && !url.includes(':5432')) {
  throw new Error(
    'scripts/db.ts: --target=prod must use the SESSION pooler (port 5432). The transaction pooler (6543) cannot run migrations.',
  );
}

const args =
  command === 'custom'
    ? ['drizzle-kit', 'generate', '--custom', ...passthrough]
    : ['drizzle-kit', command, ...passthrough];

const run = spawnSync('pnpm', ['exec', ...args], {
  stdio: 'inherit',
  shell: process.platform === 'win32',
  env: { ...process.env, DRIZZLE_DB_URL: url },
});
if (run.status !== 0) process.exit(run.status ?? 1);

// After a successful local migrate, give app_user a password so the dev server and CI
// can connect as the non-owner runtime role (D-11b). Localhost only — never the pooler.

```
-- 0000_bootstrap.sql — make a bare PostgreSQL 18 look like Supabase.
--
-- Runs in three places and must be identical in all of them: the dev machine's native
-- PostgreSQL 18 (D-05a, because Docker Desktop and WSL are both absent), CI's postgres:18
-- service container, and the real Supabase project — where these roles already exist,
-- which is why every statement is conditional.

do $$ begin
  if not exists (select 1 from pg_roles where rolname = 'anon') then
    create role anon nologin;
  end if;
  if not exists (select 1 from pg_roles where rolname = 'authenticated') then
    create role authenticated nologin;
  end if;
  if not exists (select 1 from pg_roles where rolname = 'service_role') then
    create role service_role nologin bypassrls;
  end if;
end $$;
--> statement-breakpoint

-- The runtime connects as this role, NOT as the owner. A table owner bypasses RLS, so a
-- code path that forgets `set local role` on an owner connection reads every tenant's
-- rows and nothing complains (reproduced on PostgreSQL 18.3). NOINHERIT means app_user
-- holds `authenticated` but does not use its privileges until it explicitly SET ROLEs —
-- so a forgotten `set local role` raises 42501 permission denied instead. D-11b.
do $$ begin
  if not exists (select 1 from pg_roles where rolname = 'app_user') then
    create role app_user login noinherit;
  end if;
end $$;
--> statement-breakpoint

grant authenticated to app_user;
--> statement-breakpoint

create schema if not exists app;
--> statement-breakpoint

-- Verbatim from BIS 0001_tenancy.sql:1-7. Works identically on Supabase and on bare
-- Postgres, which is exactly what makes the test suite portable.
create or replace function app.jwt() returns jsonb
language sql stable as $$
  select coalesce(nullif(current_setting('request.jwt.claims', true), '')::jsonb, '{}'::jsonb)
$$;
--> statement-breakpoint

-- Not optional. BIS shipped 0001 without these and every policy raised
-- "permission denied for schema app" instead of evaluating — which reads as an error,
-- not as a security hole, and cost a migration to find.
grant usage on schema app to authenticated, anon, service_role, app_user;
--> statement-breakpoint
grant execute on all functions in schema app to authenticated, anon, service_role;
--> statement-breakpoint
alter default privileges in schema app grant execute on functions to authenticated, anon, service_role;
--> statement-breakpoint

grant usage on schema public to authenticated, anon, service_role, app_user;
```

Do NOT put `app.current_org_id()` here. PostgreSQL validates a `language sql` function body at CREATE time, and that function selects from `public.orgs`, which does not exist until plan 05's migration. Putting it here makes `pnpm db:migrate` fail with `relation "public.orgs" does not exist` — plan 05 creates it in the same migration as the table.

Do NOT grant any table privilege directly to `app_user`. Its only path to a table is `set local role authenticated`, and that is the property the "without set local role" test in plan 05 proves.

Every statement is separated by a `--> statement-breakpoint` line on its own. Each `$$ ... $$` body stays one statement — a naive `;` splitter cuts a PL/pgSQL body in half and produces a syntax error mid-function.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| migration runner → database | `drizzle-kit migrate` runs as the owner and can alter any object; which database it reaches is decided entirely by one environment variable |
| runtime connection → tables | The role a request connects as decides whether RLS binds at all |
| SQL function → caller privileges | A `SECURITY DEFINER` function without a pinned `search_path` runs attacker-supplied code as the owner |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan | Severity |
|-----------|----------|-----------|-------------|-----------------|----------|
| T-1-03 | Elevation of Privilege | runtime database role | mitigate | `drizzle/0000_bootstrap.sql` creates `app_user login noinherit` which owns nothing and holds `authenticated` only as an un-inherited membership. A connection that forgets `set local role` gets `42501 permission denied for table`, not every tenant's rows. Task 2; proven by plan 05's "without set local role" test. | high |
| T-1-15 | Tampering | `scripts/db.ts` | mitigate | `--target=test` throws before spawning drizzle-kit if the URL matches a Supabase host; `--target=prod` throws unless it does AND uses port 5432. One file decides which database every migration reaches. Task 1. | high |
| T-1-07 | Elevation of Privilege | `app.jwt()` | mitigate | `app.jwt()` is `language sql stable` and is NOT `SECURITY DEFINER` — it needs no elevation because `current_setting` is available to every role.
